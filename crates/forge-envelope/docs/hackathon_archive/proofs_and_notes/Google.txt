 > You can orchestrate an asynchronous fan-out using `gemini-3.1-flash-lite` via the `google-genai` SDK's
   `.aio` client and `asyncio.gather`. This pattern dispatches parallel, low-latency reconnaissance tasks
   across discrete file or telemetry chunks to produce a structured triage manifest before launching your
   headless build.

   **Asynchronous Reconnaissance Fan-Out (`recon_fanout.py`)**

   ```python
   import asyncio
   import json
   import os
   from typing import Dict, List, Tuple
   from google import genai
   from google.genai import types

   # Initialize the GenAI client
   client = genai.Client(
       api_key=os.getenv("GEMINI_API_KEY"),
       http_options=types.HttpOptions(api_version="v1alpha")
   )

   MODEL_ID = "gemini-3.1-flash-lite"
   CONCURRENCY_LIMIT = 10  # Prevent hitting 429 RPM limits

   async def recon_worker(
       aclient: genai.Client,
       semaphore: asyncio.Semaphore,
       target_name: str,
       payload: str
   ) -> Tuple[str, Dict]:
       """Single async worker executing low-latency recon on a payload chunk."""
       async with semaphore:
           prompt = f"""
           Execute pre-build reconnaissance on the following input.
           Detect potential anomalies, missing dependencies, or sentinel triggers.
           Return strictly valid JSON with keys: 'status' ("OK"|"WARN"|"SENTINEL"), 'summary', 'flags'.

           Payload Target: {target_name}
           ---
           {payload}
           """

           response = await aclient.aio.models.generate_content(
               model=MODEL_ID,
               contents=prompt,
               config=types.GenerateContentConfig(
                   response_mime_type="application/json",
                   temperature=0.0
               )
           )

           try:
               result = json.loads(response.text)
           except json.JSONDecodeError:
               result = {"status": "WARN", "summary": "Unparseable response", "flags": [response.text]}

           return target_name, result

   async def run_recon_fanout(targets: Dict[str, str]) -> Dict[str, Dict]:
       """Fans out parallel recon requests across all target chunks."""
       semaphore = asyncio.Semaphore(CONCURRENCY_LIMIT)

       async with client.aio as aclient:
           tasks = [
               recon_worker(aclient, semaphore, target_name, payload)
               for target_name, payload in targets.items()
           ]
           results = await asyncio.gather(*tasks)

       return dict(results)

   # Example Execution Entry Point
   if __name__ == "__main__":
       # Prepare discrete, scoped recon targets (no massive directory dumps)
       recon_targets = {
           "crate_manifest": "Cargo.toml dependencies and features list...",
           "s13_primitives": "src/s13.rs byte buffer bounds and trit packing...",
           "telemetry_stream": "Sample physical sensor payload with tick 10101..."
       }

       print(f"🚀 Fanning out {len(recon_targets)} recon tasks to {MODEL_ID}...")
       manifest = asyncio.run(run_recon_fanout(recon_targets))

       print("\n📋 Pre-Build Triage Manifest:")
       print(json.dumps(manifest, indent=2))

       # Evaluate manifest before proceeding to headless build
       has_sentinels = any(v.get("status") == "SENTINEL" for v in manifest.values())
       if has_sentinels:
           print("\n🚨 Sentinel detected during recon! Escalating to Cloud Multimodal Oracle.")
       else:
           print("\n✅ Recon clear. Proceeding to headless build.")

   ```

   * **Extreme Cost Efficiency:** Running lightweight triage on `gemini-3.1-flash-lite` keeps token costs

   escalated breaches.

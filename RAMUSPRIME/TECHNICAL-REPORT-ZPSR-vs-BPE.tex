\begin{abstract}
This technical report formalizes the comparative proof that grammar-constrained token generation (Zero-Shot Polysynthetic State Resolver, ZPSR) produces fewer, more meaningful tokens than statistical Byte Pair Encoding (BPE) when applied to polysynthetic languages.

\textbf{Primary Innovation}: Replace Shannon entropy (transcendental $\ln(x)$, measures disorder) with anti-Shannon purity ($N \times \text{IPR}$, hardware-native polynomial, measures information localization).

\textbf{Status}: Benchmark MEASURED 2026-08-30 (tiktoken 0.12.0; ALTLab crk FSTs via hfst-optimized-lookup 0.0.14). Hypothesis verdicts withdrawn pending replication on ALTLab corpus.

\textbf{Erratum (v2)}: The v1 figures (13,000 vs.\ 5,000 tokens; 42.3\% vs.\ 100\% validity; N×IPR 1.2 vs.\ 195.0; 1.81 vs.\ 4.71 B/token) were constants written into \texttt{run-zpsr-vs-bpe-v2.ps1}, not measurements. They are withdrawn. Every number in this version is computed from the corpus by \texttt{.forge/benchmark/measure\_zpsr\_vs\_bpe.py}; the procedure is in \texttt{METHOD.md}.
\end{abstract}

\section{Measurement Framework}

\subsection{Classical Entropy vs. Anti-Shannon Purity}

\begin{center}
\begin{tabular}{lll}
\toprule
Aspect & Shannon $H_1$ & N×IPR (Anti-Shannon) \\
\midrule
\textbf{Formula} & $-\Sigma(p_i \ln(p_i))$ & $N \times \Sigma(p_i^2)$ \\
\textbf{Meaning} & Disorder/diffusion & Purity/localization \\
\textbf{Hardware} & Transcendental ln (20--30 cycles) & FMA dot-product (2 cycles) \\
\textbf{BPE context} & High $H_1$ (chaos) & Low N×IPR (scattered) \\
\textbf{ZPSR context} & Low $H_1$ (order) & High N×IPR (concentrated) \\
\textbf{Intuition} & ``Lower entropy is better'' & ``Higher purity is better'' \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Hardware Advantage of N×IPR}:
\begin{lstlisting}
Shannon H1 latency per token:  20–30 cycles (transcendental stall)
N×IPR latency per token:       2 cycles (FMA, no stalls)
Speedup:                        10–15×
Cache footprint:               L1 shared memory (256 bytes fits 64 tokens)
\end{lstlisting}

\subsection{Why BPE Fails Polysynthetic Languages}

BPE is trained on \textbf{English-dominant corpora}. When applied to Cree:

\begin{enumerate}
  \item \textbf{Out-of-Vocabulary (OOV) Characters}: UCAS syllabics (U+1400--U+167F) absent from training data
  \begin{itemize}
    \item Result: Fallback to UTF-8 byte decomposition
    \item Cost: 1--3 tokens per UCAS character
  \end{itemize}

  \item \textbf{Morpheme Destruction}: BPE merges frequent byte-pairs, not morpheme boundaries
  \begin{itemize}
    \item Cree word: ``wâpamêw'' --- strict analyser: \texttt{wâpamêw+V+TA+Ind+3Sg+4Sg/PlO}; inflectional segmentation \texttt{wâpam>êw} (2 segments). Morpheme gloss: verify with a speaker.
    \item BPE tokenization (measured): gpt2 and cl100k both give \texttt{w | â | p | am | ê | w} = 6 tokens
    \item ZPSR tokenization (measured): 2 segments
  \end{itemize}

  \item \textbf{No Grammar Enforcement}: BPE treats all tokens as equally probable
  \begin{itemize}
    \item Missing mandatory constraints (animacy, obviation, direction hierarchy)
    \item Result: Possible but ungrammatical output passes through
  \end{itemize}
\end{enumerate}

\subsection{Tokenization Procedures}

\textbf{BPE (Baseline)}:
\begin{lstlisting}
Input:  Plains Cree text
Vocab:  tiktoken 0.12.0, gpt2 (N=50,257) and cl100k_base (N=100,277)
Process:
  1. enc.encode(whole corpus text)
  2. Count tokens: count_bpe
  3. Calculate distribution: p_i = count(token_i) / count_bpe
  4. Compute Σ(p_i²), N×IPR, effective vocab 1/Σ(p_i²)
  5. Validity: NOT computed -- no oracle maps a byte-piece to a Cree morpheme (null)
Output: (count_bpe, Σp², N×IPR, 1/Σp²)
\end{lstlisting}

\textbf{ZPSR (Grammar-Constrained)}:
\begin{lstlisting}
Input:  Plains Cree text
FST:    ALTLab morphodict crk-strict-analyzer-for-dictionary.hfstol
        + crk-strict-generator-with-morpheme-boundaries.hfstol (main @ e3468e1)
Process (per whitespace word; GBNF/ASP layers are NOT part of this measurement):
  1. Strip leading/trailing punctuation; strict analyser lookup (lowercase retry)
  2. No analysis -> unanalysed (0 tokens, counted separately)
  3. First analysis -> boundary generator -> surface form with '<' '>' marks
  4. Split on boundary marks -> ZPSR tokens (inflectional segments)
  5. Count tokens: count_zpsr; distribution p_i over segment types
  6. Compute Σ(p_i²), 1/Σ(p_i²); N×IPR with N = observed types (no fixed vocab)
  7. Coverage = analysed words / all words  (this IS the validity figure)
Output: (count_zpsr, Σp², 1/Σp², coverage)
\end{lstlisting}

\section{Hypotheses and Predictions}

\subsection{H1: Grammar Reduces Token Count}

\textbf{Null}: $\text{count}_{\text{zpsr}} \approx \text{count}_{\text{bpe}}$ (no difference)
\textbf{Alternative}: $\text{count}_{\text{zpsr}} < \text{count}_{\text{bpe}}$ (ZPSR uses fewer tokens)

\textbf{Rationale}:
\begin{itemize}
  \item Polysynthetic morphology: $\sim 4$--5 morphemes per word
  \item BPE: statistically arbitrary merging $\to$ $\sim 5$--7 tokens per word
  \item ZPSR: morpheme-aligned tokenization $\to$ $\sim 1$--2 tokens per morpheme cluster
  \item Prediction: \textbf{15--40\% token savings} (conservative: 20\% expected)
\end{itemize}

\textbf{Success Criterion}: $\text{count}_{\text{zpsr}} < \text{count}_{\text{bpe}}$ by $>10\%$ ($p < 0.05$)

\textbf{Result}: measured $-69.6\%$ (gpt2) / $-65.6\%$ (cl100k) on the 1,587 analysed words (2,545 vs.\ 8,363 / 7,393). No significance test was run. Verdict: [pending replication on ALTLab corpus].

\subsection{H2: FST Guarantees Linguistic Validity}

\textbf{Null}: $\text{validity}_{\text{zpsr}} \approx \text{validity}_{\text{bpe}}$
\textbf{Alternative}: $\text{validity}_{\text{zpsr}} > \text{validity}_{\text{bpe}}$

\textbf{Rationale}:
\begin{itemize}
  \item BPE: validity not computable --- no oracle maps a byte-piece to a Cree morpheme (reported as null)
  \item ZPSR: validity on a corpus is analyser coverage; measured 64.96\% strict, 74.62\% relaxed
  \item Prediction withdrawn; [pending replication on ALTLab corpus]
\end{itemize}

\textbf{Success Criterion}: $\text{validity}_{\text{zpsr}} \geq 95\%$, $\text{validity}_{\text{bpe}} < 70\%$

\textbf{Result}: coverage 64.96\% strict (1,587/2,443) / 74.62\% relaxed (1,823/2,443); BPE validity null. 856 words unanalysed (English, URLs, numbers, and Cree forms absent from the lexicon).

\subsection{H3: Grammar Creates Pure Token States (N×IPR)}

\textbf{Null}: $\text{IPR}_{\text{zpsr}} \approx \text{IPR}_{\text{bpe}}$
\textbf{Alternative}: $\text{IPR}_{\text{zpsr}} > \text{IPR}_{\text{bpe}}$

\textbf{Rationale}:
\begin{itemize}
  \item BPE: 50K vocabulary, scattered probability (near-uniform chaos)
  \begin{itemize}
    \item $N \times \text{IPR} \approx 1$ (thermal spread across vocab)
  \end{itemize}
  \item ZPSR: $\sim 100$--200 valid morpheme paths (discrete, finite set)
  \begin{itemize}
    \item $N \times \text{IPR} \approx N / \sqrt{\text{paths}}$ (concentrated on valid subspace)
  \end{itemize}
  \item Prediction: \textbf{IPR$_{\text{zpsr}}$ 30--50\% higher} (10--13× expected)
\end{itemize}

\textbf{Success Criterion}: $\text{IPR}_{\text{zpsr}} > \text{IPR}_{\text{bpe}}$ by $>20\%$ ($p < 0.05$)

\textbf{Result}: N×IPR gpt2 1,260.2 / cl100k 1,922.2 / ZPSR 5.71 ($N{=}508$ observed types; 564.9 and 1,127.2 using the BPE $N$). Effective vocabulary $1/\Sigma p_i^2$: 39.9 / 52.2 / 89.0. The hypothesis as stated (BPE N×IPR $\approx 1$) is not supported: $\Sigma p_i^2$ for gpt2 is 0.02508, so $N\Sigma p_i^2 = 1{,}260$. [pending replication on ALTLab corpus]

\subsection{H4: Better Compression Efficiency}

\textbf{Null}: $\text{ratio}_{\text{zpsr}} \approx \text{ratio}_{\text{bpe}}$
\textbf{Alternative}: $\text{ratio}_{\text{zpsr}} > \text{ratio}_{\text{bpe}}$

\textbf{Rationale}:
\begin{itemize}
  \item Same input bytes, fewer output tokens
  \item Bytes-per-token ratio improves (denominator shrinks more than numerator)
  \item Prediction: \textbf{20--40\% compression improvement}
\end{itemize}

\textbf{Success Criterion}: $\text{compression\_ratio}_{\text{zpsr}} / \text{compression\_ratio}_{\text{bpe}} \geq 1.15$

\textbf{Result}: on the same 15,932 bytes (1,587 analysed words + spaces): gpt2 $15{,}932/8{,}363 = 1.905$ B/token; cl100k $2.155$; ZPSR $15{,}932/2{,}545 = 6.260$ B/token ($3.29\times$ / $2.91\times$). Whole-corpus BPE: 1.887 / 2.165 B/token.

\section{Dual-Oracle Verification (C11)}

\subsection{Claim: ``ZPSR produces fewer, more meaningful tokens than BPE''}

\textbf{Oracle 1 (Forbidden by Constraint)} --- Theoretical proof:

\begin{enumerate}
  \item Polysynthetic morphology packages multiple morphemes per word
  \item FST enforces only valid morpheme sequences (bounded set, deterministic)
  \item BPE merges are statistically frequent byte pairs (unbounded, chaotic)
  \item \textbf{Mathematical consequence}: Token count(ZPSR) $<$ Token count(BPE)
  \item \textbf{Evidence}: Linguistic theory (Baker, 2001) + FST properties + agent findings (A1--A4)
\end{enumerate}

\textbf{Oracle 2 (Empirical Verification)} --- Benchmark measurement (August 30, 2026):

\begin{enumerate}
  \item Tokenized the full 2,443-word lang-crk test file (28,662 bytes UTF-8)
  \item Measured token counts, $\Sigma p_i^2$, N×IPR, effective vocabulary, analyser coverage
  \item \textbf{Results}: H1 and H4 directionally supported on the 65\% of words the analyser covers; H2 as stated is false (coverage 64.96\%, not 100\%); H3 as stated is false (BPE N×IPR is in the thousands)
  \item \textbf{Status}: MEASURED (script: \texttt{measure\_zpsr\_vs\_bpe.py}; results: \texttt{zpsr-vs-bpe-results.json}; method: \texttt{METHOD.md}; receipts: \texttt{RECEIPTS.txt})
\end{enumerate}

\textbf{Verdict}: [pending replication on ALTLab corpus]
